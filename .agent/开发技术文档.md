## 三、主进程架构

### 3.1 启动流程

```
app.whenReady()
  │
  ├─ electronApp.setAppUserModelId('com.kuang-lan')
  ├─ optimizer.watchWindowShortcuts(window)
  │
  ├─ initDatabase()                    # 1. 初始化数据库
  │   ├─ 检查 userData/kuang_lan.db 是否存在
  │   ├─ 不存在 → 创建数据库 + 执行 schema + 插入默认分组
  │   └─ 已存在 → 校验 schema_version，必要时迁移
  │
  ├─ importPresets()                   # 2. 扫描并导入预设
  │   ├─ 读取 resources/presets/*.json
  │   ├─ 首次启动 → 全量导入
  │   └─ 非首次 → 仅导入新增预设
  │
  ├─ registerIpcHandlers()             # 3. 注册 IPC handlers
  │
  ├─ createWindow()                    # 4. 创建主窗口
  │   ├─ 读取设置中的窗口 bounds
  │   ├─ new BrowserWindow({...})
  │   └─ 加载 renderer
  │
  └─ mainWindow.on('ready-to-show')
      ├─ mainWindow.show()
      └─ mainWindow.webContents.send('db:init', loadData())
          # 5. 推送初始数据到渲染进程
```

### 3.2 主进程入口 `src/main/index.ts`

```ts
import { app, shell, BrowserWindow, ipcMain, nativeTheme } from 'electron'
import { join } from 'path'
import { electronApp, optimizer, is } from '@electron-toolkit/utils'
import { initDatabase, loadData } from './database'
import { importPresets } from './preset'
import { registerIpcHandlers } from './ipc'

let mainWindow: BrowserWindow | null = null

function createWindow(): void {
  mainWindow = new BrowserWindow({
    width: 1200,
    height: 800,
    minWidth: 800,
    minHeight: 600,
    show: false,
    autoHideMenuBar: true,
    frame: false,                       // 无边框，自定义标题栏
    titleBarStyle: 'hidden',            // macOS 保留红绿灯
    webPreferences: {
      preload: join(__dirname, '../preload/index.js'),
      sandbox: false                    // better-sqlite3 需要
    }
  })

  mainWindow.on('ready-to-show', () => {
    mainWindow!.show()
    // 推送初始数据
    mainWindow!.webContents.send('db:init', loadData())
  })

  mainWindow.webContents.setWindowOpenHandler((details) => {
    shell.openExternal(details.url)
    return { action: 'deny' }
  })

  if (is.dev && process.env['ELECTRON_RENDERER_URL']) {
    mainWindow.loadURL(process.env['ELECTRON_RENDERER_URL'])
  } else {
    mainWindow.loadFile(join(__dirname, '../renderer/index.html'))
  }
}

app.whenReady().then(() => {
  electronApp.setAppUserModelId('com.kuang-lan')

  app.on('browser-window-created', (_, window) => {
    optimizer.watchWindowShortcuts(window)
  })

  // 按顺序初始化
  initDatabase()
  importPresets()
  registerIpcHandlers()
  createWindow()

  app.on('activate', () => {
    if (BrowserWindow.getAllWindows().length === 0) createWindow()
  })
})

app.on('window-all-closed', () => {
  if (process.platform !== 'darwin') app.quit()
})
```

### 3.3 数据库模块

#### `src/main/database/index.ts`

```ts
import Database from 'better-sqlite3'
import { app } from 'electron'
import { join } from 'path'
import { createSchema } from './schema'

const DB_NAME = 'kuang_lan.db'
let db: Database.Database | null = null

export function initDatabase(): void {
  const dbPath = join(app.getPath('userData'), DB_NAME)
  db = new Database(dbPath)

  // 启用 WAL 模式，提升并发读性能
  db.pragma('journal_mode = WAL')
  db.pragma('foreign_keys = ON')

  // 检查是否需要初始化
  const tableExists = db
    .prepare("SELECT name FROM sqlite_master WHERE type='table' AND name='schema_version'")
    .get()

  if (!tableExists) {
    createSchema(db)
  } else {
    migrateIfNeeded(db)
  }
}

export function getDb(): Database.Database {
  if (!db) throw new Error('Database not initialized')
  return db
}

export function loadData(): { groups: Group[]; views: View[]; settings: Record<string, string> } {
  const d = getDb()
  const groups = d.prepare('SELECT * FROM groups ORDER BY sort_order').all()
  const views = d.prepare('SELECT * FROM views ORDER BY sort_order').all()
  const settingsRows = d.prepare('SELECT key, value FROM settings').all()
  const settings: Record<string, string> = {}
  for (const row of settingsRows) {
    settings[row.key] = row.value
  }
  return { groups, views, settings }
}
```

```ts
import Database from 'better-sqlite3'
import { v4 as uuidv4 } from 'uuid'

const CURRENT_VERSION = 1

export function createSchema(db: Database.Database): void {
  db.exec(`
    CREATE TABLE groups (
      id          TEXT PRIMARY KEY,
      name        TEXT NOT NULL,
      icon        TEXT NOT NULL DEFAULT '📁',
      color       TEXT NOT NULL DEFAULT '#FF8FA3',
      sort_order  INTEGER NOT NULL DEFAULT 0,
      collapsed   INTEGER NOT NULL DEFAULT 0,
      is_default  INTEGER NOT NULL DEFAULT 0,
      created_at  TEXT NOT NULL DEFAULT (datetime('now')),
      updated_at  TEXT NOT NULL DEFAULT (datetime('now'))
    );

    CREATE TABLE views (
      id          TEXT PRIMARY KEY,
      group_id    TEXT NOT NULL REFERENCES groups(id) ON DELETE SET NULL,
      name        TEXT NOT NULL,
      url         TEXT NOT NULL,
      icon        TEXT DEFAULT '',
      visible     INTEGER NOT NULL DEFAULT 1,
      sort_order  INTEGER NOT NULL DEFAULT 0,
      bounds      TEXT DEFAULT '{}',
      created_at  TEXT NOT NULL DEFAULT (datetime('now')),
      updated_at  TEXT NOT NULL DEFAULT (datetime('now'))
    );

    CREATE TABLE settings (
      key         TEXT PRIMARY KEY,
      value       TEXT NOT NULL,
      updated_at  TEXT NOT NULL DEFAULT (datetime('now'))
    );

    CREATE TABLE schema_version (
      version     INTEGER PRIMARY KEY,
      applied_at  TEXT NOT NULL DEFAULT (datetime('now'))
    );

    CREATE INDEX idx_views_group_id ON views(group_id);
    CREATE INDEX idx_views_sort_order ON views(sort_order);
    CREATE INDEX idx_groups_sort_order ON groups(sort_order);
  `)

  // 插入默认分组
  db.prepare(`
    INSERT INTO groups (id, name, icon, color, sort_order, is_default)
    VALUES (?, '默认分组', '📁', '#FF8FA3', 0, 1)
  `).run(uuidv4())

  // 记录版本
  db.prepare('INSERT INTO schema_version (version) VALUES (?)').run(CURRENT_VERSION)
}

export function migrateIfNeeded(db: Database.Database): void {
  const row = db.prepare('SELECT MAX(version) as v FROM schema_version').get() as { v: number }
  const current = row.v

  if (current < CURRENT_VERSION) {
    // 依次执行迁移脚本
    for (let v = current + 1; v <= CURRENT_VERSION; v++) {
      const migration = getMigration(v)
      if (migration) {
        db.exec(migration)
        db.prepare('INSERT INTO schema_version (version) VALUES (?)').run(v)
      }
    }
  }
}

function getMigration(version: number): string | null {
  // 预留迁移脚本
  // switch (version) {
  //   case 2: return `ALTER TABLE views ADD COLUMN tags TEXT DEFAULT '[]';`
  // }
  return null
}
```

#### `src/main/database/group-repo.ts`

```ts
import { getDb } from './index'
import { v4 as uuidv4 } from 'uuid'

export interface Group {
  id: string
  name: string
  icon: string
  color: string
  sort_order: number
  collapsed: number
  is_default: number
  created_at: string
  updated_at: string
}

export function createGroup(data: Omit<Group, 'id' | 'created_at' | 'updated_at'>): Group {
  const db = getDb()
  const id = uuidv4()
  const now = new Date().toISOString()
  db.prepare(`
    INSERT INTO groups (id, name, icon, color, sort_order, collapsed, is_default, created_at, updated_at)
    VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
  `).run(id, data.name, data.icon, data.color, data.sort_order, data.collapsed, data.is_default ?? 0, now, now)
  return db.prepare('SELECT * FROM groups WHERE id = ?').get(id) as Group
}

export function updateGroup(id: string, data: Partial<Omit<Group, 'id' | 'created_at'>>): Group {
  const db = getDb()
  const fields: string[] = []
  const values: unknown[] = []

  for (const [key, value] of Object.entries(data)) {
    if (key === 'id' || key === 'created_at') continue
    fields.push(`${key} = ?`)
    values.push(value)
  }

  fields.push('updated_at = ?')
  values.push(new Date().toISOString())
  values.push(id)

  db.prepare(`UPDATE groups SET ${fields.join(', ')} WHERE id = ?`).run(...values)
  return db.prepare('SELECT * FROM groups WHERE id = ?').get(id) as Group
}

export function deleteGroup(id: string): void {
  const db = getDb()
  // 将该分组下的视图移至默认分组
  const defaultGroup = db.prepare("SELECT id FROM groups WHERE is_default = 1").get() as { id: string }
  if (defaultGroup) {
    db.prepare('UPDATE views SET group_id = ? WHERE group_id = ?').run(defaultGroup.id, id)
  }
  db.prepare('DELETE FROM groups WHERE id = ? AND is_default = 0').run(id)
}

export function reorderGroup(id: string, sortOrder: number): void {
  const db = getDb()
  db.prepare('UPDATE groups SET sort_order = ?, updated_at = ? WHERE id = ?')
    .run(sortOrder, new Date().toISOString(), id)
}

export function getAllGroups(): Group[] {
  return getDb().prepare('SELECT * FROM groups ORDER BY sort_order').all() as Group[]
}
```

#### `src/main/preset/index.ts`

```ts
import { app } from 'electron'
import { join } from 'path'
import { readdirSync, readFileSync, existsSync } from 'fs'
import { getDb } from '../database'
import { createGroup } from '../database/group-repo'
import { createView } from '../database/view-repo'
import { setSetting } from '../database/setting-repo'
import type { PresetFile } from './types'

export function importPresets(): void {
  const db = getDb()

  // 判断是否首次启动（groups 表只有默认分组）
  const groupCount = (db.prepare('SELECT COUNT(*) as c FROM groups').get() as { c: number }).c
  const isFirstLaunch = groupCount <= 1

  // 预设目录：开发时从 resources/presets 读取，打包后从 process.resourcesPath 读取
  const presetsDir = is.dev
    ? join(__dirname, '../../resources/presets')
    : join(process.resourcesPath, 'presets')

  if (!existsSync(presetsDir)) return

  const files = readdirSync(presetsDir).filter((f) => f.endsWith('.json'))

  for (const file of files) {
    try {
      const content = readFileSync(join(presetsDir, file), 'utf-8')
      const preset: PresetFile = JSON.parse(content)

      if (!validatePreset(preset)) continue

      // 检查是否已导入（按 group.name 去重）
      const existing = db.prepare('SELECT id FROM groups WHERE name = ?').get(preset.group.name)
      if (existing && !isFirstLaunch) continue

      // 如果已存在但首次启动，跳过（说明之前已导入过）
      if (existing) continue

      // 创建分组
      const group = createGroup({
        name: preset.group.name,
        icon: preset.group.icon,
        color: preset.group.color,
        sort_order: 0,
        collapsed: 0,
        is_default: 0
      })

      // 创建视图
      for (let i = 0; i < preset.views.length; i++) {
        const v = preset.views[i]
        createView({
          group_id: group.id,
          name: v.name,
          url: v.url,
          icon: v.icon,
          visible: 1,
          sort_order: i,
          bounds: '{}'
        })
      }

      // 记录已导入的预设文件
      setSetting(`preset_imported_${file}`, new Date().toISOString())
    } catch (err) {
      console.error(`Failed to import preset ${file}:`, err)
    }
  }
}

function validatePreset(preset: PresetFile): boolean {
  return (
    preset.version === 1 &&
    typeof preset.group?.name === 'string' &&
    Array.isArray(preset.views) &&
    preset.views.every((v) => v.name && v.url)
  )
}
```

```ts
import { ref, onMounted, onUnmounted } from 'vue'

type Theme = 'light' | 'dark'

const currentTheme = ref<Theme>('light')

export function useTheme() {
  function applyTheme(theme: Theme) {
    currentTheme.value = theme
    document.documentElement.setAttribute('data-theme', theme)

    // 添加过渡类
    document.documentElement.classList.add('theme-transition')
    setTimeout(() => {
      document.documentElement.classList.remove('theme-transition')
    }, 350)

    // 持久化
    window.api.db.setting.set('theme', theme)
  }

  async function initTheme() {
    // 1. 读取用户存储偏好
    const saved = await window.api.db.setting.get('theme')
    if (saved === 'light' || saved === 'dark') {
      applyTheme(saved)
      return
    }

    // 2. 读取系统偏好
    const prefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches
    applyTheme(prefersDark ? 'dark' : 'light')
  }

  function toggleTheme() {
    applyTheme(currentTheme.value === 'light' ? 'dark' : 'light')
  }

  // 监听系统主题变化
  const mediaQuery = window.matchMedia('(prefers-color-scheme: dark)')
  function onSystemChange(e: MediaQueryListEvent) {
    // 仅在用户未手动设置时跟随系统
    window.api.db.setting.get('theme').then((saved) => {
      if (!saved) {
        applyTheme(e.matches ? 'dark' : 'light')
      }
    })
  }

  onMounted(() => {
    initTheme()
    mediaQuery.addEventListener('change', onSystemChange)
  })

  onUnmounted(() => {
    mediaQuery.removeEventListener('change', onSystemChange)
  })

  return { currentTheme, applyTheme, toggleTheme, initTheme }
}
```

```ts
export const useGroupStore = defineStore('group', () => {
  const groups = ref<Group[]>([])

  const sortedGroups = computed(() =>
    [...groups.value].sort((a, b) => a.sort_order - b.sort_order)
  )

  const defaultGroup = computed(() =>
    groups.value.find((g) => g.is_default)
  )

  function setGroups(data: Group[]) {
    groups.value = data
  }

  async function createGroup(data: Omit<Group, 'id' | 'created_at' | 'updated_at'>) {
    const group = await window.api.db.group.create(data)
    groups.value.push(group)
    return group
  }

  async function updateGroup(id: string, data: Partial<Group>) {
    const updated = await window.api.db.group.update({ id, ...data })
    const idx = groups.value.findIndex((g) => g.id === id)
    if (idx !== -1) groups.value[idx] = updated
    return updated
  }

  async function deleteGroup(id: string) {
    await window.api.db.group.delete(id)
    groups.value = groups.value.filter((g) => g.id !== id)
  }

  async function toggleCollapse(id: string) {
    const group = groups.value.find((g) => g.id === id)
    if (group) {
      await updateGroup(id, { collapsed: group.collapsed ? 0 : 1 })
    }
  }

  return { groups, sortedGroups, defaultGroup, setGroups, createGroup, updateGroup, deleteGroup, toggleCollapse }
})
```

```ts
    views.value.find((v) => v.id === activeViewId.value)
  )

    views.value.filter((v) => v.visible)
  )

  function getViewsByGroup(groupId: string) {
    return views.value
      .filter((v) => v.group_id === groupId)
      .sort((a, b) => a.sort_order - b.sort_order)
  }

  function setViews(data: View[]) {
    views.value = data
  }

  function setActiveView(id: string) {
    activeViewId.value = id
    window.api.db.setting.set('activeViewId', id)
  }

  async function createView(data: Omit<View, 'id' | 'created_at' | 'updated_at'>) {
    const view = await window.api.db.view.create(data)
    views.value.push(view)
    return view
  }

  async function updateView(id: string, data: Partial<View>) {
    const updated = await window.api.db.view.update({ id, ...data })
    const idx = views.value.findIndex((v) => v.id === id)

    return updated
  }

  async function deleteView(id: string) {
    await window.api.db.view.delete(id)
    views.value = views.value.filter((v) => v.id !== id)
    if (activeViewId.value === id) {
      activeViewId.value = views.value[0]?.id ?? null
    }
  }

  async function moveView(id: string, targetGroupId: string) {
    await window.api.db.view.move(id, targetGroupId)

    if (view) view.group_id = targetGroupId
  }

    views, activeViewId, activeView, visibleViews,
    getViewsByGroup, setViews, setActiveView, createView, updateView, deleteView, moveView
  }
```

## 六、WebView 集成方案

### 6.1 方案对比

| 方案 | 优点 | 缺点 | 推荐度 |
|---|---|---|---|
| `<webview>` 标签 | 简单，渲染进程内嵌，支持 CSS 布局 | 性能略差，需启用 webviewTag | ★★★ |
| `BrowserView` | 独立进程，性能好 | 已废弃（Electron 22+），布局需手动计算 | ★★ |
| 多 `BrowserWindow` | 完全隔离，性能最佳 | 窗口管理复杂，无法 CSS 布局 | ★★ |
| WebContentsView (Electron 28+) | 官方推荐替代 BrowserView | API 较新，文档较少 | ★★★★ |

### 6.2 推荐方案：WebContentsView

```ts
// src/main/webview-manager.ts
import { WebContentsView, BrowserWindow } from 'electron'
import type { View } from './database/view-repo'

const viewMap = new Map<string, WebContentsView>()

export function createWebView(mainWindow: BrowserWindow, view: View): WebContentsView {
  const wcView = new WebContentsView()
  wcView.webContents.loadURL(view.url)

  // 设置初始 bounds
  const bounds = mainWindow.getBounds()
  const sidebarWidth = 72
  const topbarHeight = 40
  wcView.setBounds({
    x: sidebarWidth + 16,
    y: topbarHeight + 16,
    width: bounds.width - sidebarWidth - 32,
    height: bounds.height - topbarHeight - 32
  })

  mainWindow.contentView.addChildView(wcView)
  viewMap.set(view.id, wcView)
  return wcView
}

export function showView(mainWindow: BrowserWindow, viewId: string): void {
  for (const [id, wcView] of viewMap) {
    wcView.setVisible(id === viewId)
  }
}

export function removeView(mainWindow: BrowserWindow, viewId: string): void {
  const wcView = viewMap.get(viewId)
  if (wcView) {
    mainWindow.contentView.removeChildView(wcView)
    wcView.webContents.close()
    viewMap.delete(viewId)
  }
}

export function resizeViews(mainWindow: BrowserWindow): void {
  const bounds = mainWindow.getBounds()
  const sidebarWidth = 72
  const topbarHeight = 40

  for (const wcView of viewMap.values()) {
    wcView.setBounds({
      x: sidebarWidth + 16,
      y: topbarHeight + 16,
      width: bounds.width - sidebarWidth - 32,
      height: bounds.height - topbarHeight - 32
    })
  }
}
```

主进程窗口创建时需添加：

```ts
// 在 createWindow() 中
mainWindow.on('resize', () => resizeViews(mainWindow))

// IPC：切换视图
ipcMain.handle('view:switch', (_e, viewId: string) => {
  showView(mainWindow!, viewId)
})

// IPC：新建视图
ipcMain.handle('view:create-webview', (_e, viewData: View) => {
  createWebView(mainWindow!, viewData)
})

// IPC：刷新视图
ipcMain.handle('view:reload', (_e, viewId: string) => {
  const wcView = viewMap.get(viewId)
  if (wcView) wcView.webContents.reload()
})
```
